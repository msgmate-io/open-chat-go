// Package benchmark implements the reproducible, machine-readable harness for
// the "free OpenRouter model x private sandbox test matrix" capability report
// (msgmate-io/open-chat-go-ci#893).
//
// Design rules enforced here:
//   - The free-model input list is derived from the model registry files on
//     disk (LoadFreeModels), never hardcoded and never fetched from a live
//     API. See registry.go.
//   - Result aggregation and table rendering are pure functions over local
//     fixtures/TSVs; they never touch the network.
//   - The expensive full-matrix sweep is NOT a unit test. It is launched only
//     by the explicitly-triggered CLI subcommand `dispatch` in
//     ./cmd/benchmark, which hands off to the existing private workflow
//     .github/workflows/private-sandbox-tests.yaml (see
//     development/ci893-free-model-capability-harness.md).
package benchmark

import (
	"fmt"
	"time"
)

// StatusCategory is the coarse classification the capability table renders.
// provider_* and timeout are deliberately kept distinct from genuine
// capability failures, matching the ci#479 provider-failure classification used
// by the private runner.
type StatusCategory string

const (
	CategoryPass           StatusCategory = "pass"
	CategoryFail           StatusCategory = "fail"
	CategoryFailAfterRetry StatusCategory = "fail_after_retry"
	CategoryProviderError  StatusCategory = "provider_error"
	CategoryTimeout        StatusCategory = "timeout"
	CategorySkipped        StatusCategory = "skipped"
	CategoryNotApplicable  StatusCategory = "n/a"
	CategoryNotRun         StatusCategory = "not_run"
)

// SchemaVersion is the version of the emitted machine-readable artifact
// (benchmark-results/free-models-<date>.json).
const SchemaVersion = 1

// FreeModel is one model from the registry that the sweep should target.
// Only the model identifier is acted upon; everything else is provenance.
type FreeModel struct {
	ID     string `json:"model_id"`
	Title  string `json:"title,omitempty"`
	Source string `json:"registry_source,omitempty"` // file the entry came from
}

// TestResult is the aggregated final outcome of one (model, test) pair.
type TestResult struct {
	TestID   string         `json:"test_id"`
	Tag      string         `json:"tag,omitempty"`
	Status   string         `json:"status"`   // raw final status as recorded (e.g. "provider_rate_limit")
	Category StatusCategory `json:"category"` // coarse class for the table
	Reason   string         `json:"reason,omitempty"`
	Attempts int            `json:"attempts"`
}

// Counts holds the per-model / per-test tally used for the pass-rate column.
type Counts struct {
	Passed         int `json:"passed"`
	Failed         int `json:"failed"`
	FailAfterRetry int `json:"fail_after_retry"`
	ProviderError  int `json:"provider_error"`
	Timeout        int `json:"timeout"`
	Skipped        int `json:"skipped"`
	NotApplicable  int `json:"not_applicable"`
	NotRun         int `json:"not_run"`
}

// Total counts the outcomes that participate in the pass-rate denominator:
// pass, fail, provider-error and timeout are real results; skipped / n/a /
// not-run are documented as absences, not capabilities.
func (c Counts) Total() int {
	return c.Passed + c.Failed + c.FailAfterRetry + c.ProviderError + c.Timeout
}

// PassRate returns "NN.N%" over real outcomes, or "n/a" when there are none.
func (c Counts) PassRate() string {
	if c.Total() == 0 {
		return "n/a"
	}
	pct := float64(c.Passed) / float64(c.Total()) * 100
	return formatRate(pct)
}

// ModelReport is one row of the capability table.
type ModelReport struct {
	ModelID  string       `json:"model_id"`
	Title    string       `json:"title,omitempty"`
	Registry string       `json:"registry_source,omitempty"`
	Counts   Counts       `json:"counts"`
	PassRate string       `json:"pass_rate"`
	Results  []TestResult `json:"results"`
}

// TestSummary is one column of the capability table (per-test pass rate).
type TestSummary struct {
	TestID   string `json:"test_id"`
	Tag      string `json:"tag,omitempty"`
	Counts   Counts `json:"counts"`
	PassRate string `json:"pass_rate"`
}

// MatrixReport is the machine-readable artifact the landing page consumes.
type MatrixReport struct {
	SchemaVersion  int           `json:"schema_version"`
	GeneratedAt    string        `json:"generated_at"`
	TargetRef      string        `json:"target_ref,omitempty"`
	SourceRegistry []string      `json:"source_registry"`
	Models         []ModelReport `json:"models"`
	Tests          []TestSummary `json:"tests"`
}

// NewMatrixReport seeds a report with the current timestamp (UTC RFC3339).
func NewMatrixReport(targetRef string, sources []string) *MatrixReport {
	return &MatrixReport{
		SchemaVersion:  SchemaVersion,
		GeneratedAt:    time.Now().UTC().Format(time.RFC3339),
		TargetRef:      targetRef,
		SourceRegistry: sources,
		Models:         []ModelReport{},
		Tests:          []TestSummary{},
	}
}

func formatRate(pct float64) string {
	return fmt.Sprintf("%.1f%%", pct)
}

package benchmark

import (
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

// registrySources are the model-registry files the free-model set is derived
// from. They are read relative to the backend module root, mirroring how
// database/model_config.go embeds and seeds the same files.
var registrySources = []string{
	"database/data/default_model_configs.json",
	"database/data/extra_default_models_config.json",
}

// entryConfiguration is the registry entry shape used by the seeding code
// (see modelConfigFileEntry in database/model_config.go). Only the fields the
// runner needs are defined; the rest pass through untouched.
type entryConfiguration struct {
	Model   string `json:"model"`
	Backend string `json:"backend"`
	// The `tier` field is a registry-level hint (e.g. "free" / "paid") parsed
	// for documentation only; the authoritative free marker is the ":free"
	// suffix OpenRouter assigns to free-tier model identifiers.
	Tier string `json:"tier"`
}

type registryEntry struct {
	Title         string             `json:"title"`
	Configuration entryConfiguration `json:"configuration"`
}

// LoadFreeModels returns the deduplicated set of free models found in the
// registry files, in registry order. A model is considered free when its
// configuration model identifier ends with the OpenRouter free-tier marker
// ":free". The list is always read from the registry on disk so the sweep
// never hardcodes (or guesses) the set, and it never calls a live API.
func LoadFreeModels(backendRoot string) ([]FreeModel, error) {
	seen := map[string]bool{}
	var out []FreeModel
	for _, rel := range registrySources {
		path := filepath.Join(backendRoot, rel)
		data, err := os.ReadFile(path)
		if err != nil {
			return nil, fmt.Errorf("benchmark: read registry %s: %w", rel, err)
		}
		var entries []registryEntry
		if err := json.Unmarshal(data, &entries); err != nil {
			return nil, fmt.Errorf("benchmark: parse registry %s: %w", rel, err)
		}
		for _, e := range entries {
			raw := strings.ToLower(e.Configuration.Model)
			if !strings.HasSuffix(raw, ":free") {
				continue
			}
			id := e.Configuration.Model
			if seen[id] {
				continue
			}
			seen[id] = true
			out = append(out, FreeModel{ID: id, Title: e.Title, Source: rel})
		}
	}
	sort.Slice(out, func(i, j int) bool {
		return out[i].ID < out[j].ID
	})
	if len(out) == 0 {
		return nil, fmt.Errorf("benchmark: no :free models found in %v", registrySources)
	}
	return out, nil
}

// SourceRegistryNames returns the registry names for artifact provenance.
func SourceRegistryNames() []string {
	return append([]string(nil), registrySources...)
}

package benchmark

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

// AggregateTSVToJSON maps per-model result directories (named with the harness
// slug convention provider_model_free) to {modelID: {test: status}} rows.
//
// Each result directory contains one or more *.tsv files emitted by the
// private-sandbox runner; every line is "<test>\t<status>" and the last
// occurrence of a test within a model wins. Directories without result files
// are skipped, and an empty tree yields an empty map (never an error).
//
// The model identifier is reconstructed from the directory slug by converting
// underscores between the provider and model segments back to "/" and the
// trailing "_free" back to ":free". It is a pure function over the local
// benchmark-results tree; it performs no network or sandbox access, matching
// the design rules in types.go.
func AggregateTSVToJSON(resultsRoot string) (map[string]map[string]string, error) {
	rows := map[string]map[string]string{}

	modelDirs, err := os.ReadDir(resultsRoot)
	if err != nil {
		return nil, fmt.Errorf("benchmark: read results root %s: %w", resultsRoot, err)
	}

	for _, dirEntry := range modelDirs {
		if !dirEntry.IsDir() {
			continue
		}
		slug := dirEntry.Name()
		modelID := modelIDFromSlug(slug)

		statusByTest := map[string]string{}
		tsvFiles, err := filepath.Glob(filepath.Join(resultsRoot, slug, "*.tsv"))
		if err != nil {
			return nil, fmt.Errorf("benchmark: glob %s: %w", slug, err)
		}
		sort.Strings(tsvFiles)
		for _, tsv := range tsvFiles {
			lines, err := readTSVLines(tsv)
			if err != nil {
				return nil, err
			}
			for _, line := range lines {
				fields := strings.SplitN(line, "\t", 2)
				if len(fields) != 2 {
					continue
				}
				test := strings.TrimSpace(fields[0])
				status := strings.TrimSpace(fields[1])
				if test == "" || status == "" {
					continue
				}
				// Later files/rows override earlier ones so a retried or
				// finalized status wins over an intermediate one.
				statusByTest[test] = status
			}
		}
		if len(statusByTest) > 0 {
			rows[modelID] = statusByTest
		}
	}

	return rows, nil
}

// modelIDFromSlug reverses the harness slug convention. The slug is built from
// "provider/model:free" by replacing "/" with "_" and ":" with "_", so we
// restore the identifier by replacing the first "_" with "/" and the trailing
// "_free" with ":free".
func modelIDFromSlug(slug string) string {
	id := slug
	if strings.HasSuffix(id, "_free") {
		id = strings.TrimSuffix(id, "_free") + ":free"
	}
	if i := strings.IndexByte(id, '_'); i >= 0 {
		id = id[:i] + "/" + id[i+1:]
	}
	return id
}

func readTSVLines(path string) ([]string, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("benchmark: read %s: %w", path, err)
	}
	text := strings.ReplaceAll(string(data), "\r\n", "\n")
	var out []string
	for _, line := range strings.Split(text, "\n") {
		line = strings.TrimSpace(line)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		out = append(out, line)
	}
	return out, nil
}
