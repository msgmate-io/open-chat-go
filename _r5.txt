// RequestAutoApprove dispatches the ci#538 auto-approve workflow for the PR, so
// the required approving review is posted by the reviewer identity once the
// required PR CI run succeeded.
func (c *GitHubAPIClient) RequestAutoApprove(ctx context.Context, number int) error {
	payload := map[string]interface{}{
		"ref": "main",
		"inputs": map[string]string{
			"pr_number": strconv.Itoa(number),
		},
	}
	path := fmt.Sprintf("/repos/%s/actions/workflows/auto-approve-consolidation.yaml/dispatches", c.Repo)
	return c.request(ctx, http.MethodPost, path, payload, nil)
}

     9	// capabilities.go adds the small, optional extension the CI-error relay needs
    10	// (ci#607) on top of the base integration interface, without touching the
    11	// external go-integration-interface module and without forcing every

    13	//
    14	// An integration that wants to consume relayed events (today: `ci_failure`)
    15	// registers an EventSink under its own name from its bootstrap/route
    16	// registration. Integrations that do not implement the capability are
    17	// completely unaffected: the sink map stays empty and lookups simply miss, so
    18	// this extension is backwards compatible by construction.
    19	//
    20	// The contract is deliberately narrow and secret free:
    21	//
    22	//	type EventSink interface {
    23	//	    HandleIntegrationEvent(ctx context.Context, kind string, payload json.RawMessage) (any, error)
    24	//	}
    25	//
    26	// A sink receives the *raw* kind-specific payload plus the already-lower-cased
    27	// kind and owns the interpretation. Routing, retries and rendering therefore
    28	// stay in the consumer (the agent chooser / development lead), while external
    29	// systems (CI) only emit a payload.
    30	
    31	// EventSink is the optional capability an integration may provide to consume
    32	// externally relayed, versioned integration events.
    33	type EventSink interface {
    34		HandleIntegrationEvent(ctx context.Context, kind string, payload json.RawMessage) (any, error)
    35	}
