   156	// providerRetryBackoffDelay computes the exponential back-off delay before
   157	// retry attempt+1. attempt is 1-based (the delay after the first failure).
   158	func providerRetryBackoffDelay(attempt int, base, max time.Duration) time.Duration {
   159		if attempt < 1 {
   160			attempt = 1
   161		}
   162		if base <= 0 {
   163			base = 2 * time.Second
   164		}
   165		if attempt > 20 {
   166			attempt = 1
   167		}
   168		delay := base * time.Duration(1<<(attempt-1))
   169		if max > 0 && delay > max {
   170			delay = max
   171		}
   172		return delay
   173	}
